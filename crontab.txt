# Montana Blotter cron jobs
#
# Strategy (2026-06-11): re-enabled the four high-ROI workers that drive
# arrests / jail rosters / blog content, but throttled + low-priority so the
# web app keeps headroom. Each new entry is wrapped in `nice -n 19 ionice -c 3`
# and offset on the clock from the existing 15-minute email jobs to avoid
# stacking. Watchdog is kept in sync via services.ops.cron_introspection.

SHELL=/bin/bash
BASH_ENV=/root/montanablotter/.env

# Web server healthcheck — lightweight liveness probe.
*/3 * * * * /root/montanablotter/scripts/ops/healthcheck_restart.sh

# Email queue scan — polls IONOS inbox (records@montanablotter.com) every 15 min
# and routes Havre PD daily jail roster DOCX attachments to the jail_bookings
# pipeline via services/ingestion/fetchers/havre_inmate.py. Logs to mail.log.
# Shared-lock wait matches the 900s child timeout: historical successful scans
# averaged ~591s and peaked at ~891s, so shorter waits starved sibling jobs.
*/15 * * * * /root/montanablotter/venv/bin/python3 /root/montanablotter/job_runner.py --name email_worker --shared-lock blotter_db_writer --wait-for-lock 900 --timeout 900 --log /root/montanablotter/logs/mail.log --workdir /root/montanablotter -- /root/montanablotter/venv/bin/python3 /root/montanablotter/email_worker.py --mode queue

# Image-based blotter ingestion — separate from the queue scan above because
# it processes a short inbox window for any image attachments missed by the
# queue poll (Havre PDF blotters, etc.). A 2-day look-back is enough of a
# safety net because email_worker polls UNSEEN every 15 minutes. Offset by
# 8 min to avoid stacking.
8-59/15 * * * * /root/montanablotter/venv/bin/python3 /root/montanablotter/job_runner.py --name email_image_blotter --shared-lock blotter_db_writer --wait-for-lock 900 --timeout 900 --log /root/montanablotter/logs/email_image_blotter.log --workdir /root/montanablotter -- /root/montanablotter/venv/bin/python3 /root/montanablotter/email_image_blotter.py --since-days 2
